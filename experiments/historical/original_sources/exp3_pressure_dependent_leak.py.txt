"""Historical S8 computational source; original input locations are listed in dependencies.json."""
import os, sys, json, time, copy, pickle, random, warnings, tempfile, uuid
import numpy as np

warnings.filterwarnings('ignore')
os.environ.setdefault('KMP_DUPLICATE_LIB_OK', 'TRUE')
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import wntr
from scipy.stats import binomtest

_P = os.path.join(tempfile.mkdtemp(prefix='epanet_'), 'ep_' + uuid.uuid4().hex[:8])
KB, INP = 'knowledge_base', 'dataset/Exa7.inp'
PART_PKL, K = 'partition_results_leiden/partitions.pkl', 15
NOISE, SEED, PER = 0.02, 7, 'day_normal'
Q_REF = 20.0                       # reference discharge for corpus templates (L/s)
ALPHAS = [0.5, 1.0, 1.5]
TARGETS = [5.0, 20.0]
INVAR_TARGETS = [2.0, 5.0, 10.0, 20.0, 35.0, 50.0]
DENSE_CACHE, OUT = 'exp2a_dense_corpus.npz', 'exp3_results.json'


def proj_res_mat(y, F, Fn):
    b = (F @ y) / np.maximum(Fn ** 2, 1e-18)
    r = np.linalg.norm(y[None, :] - b[:, None] * F, axis=1)
    r[Fn == 0] = np.inf
    return r


def main():
    from phase3_query.leak_locator import LeakLocator
    print('=' * 90)
    print('EXPERIMENT 3 — pressure-dependent leakage: does the scale invariance survive?')
    print('=' * 90)

    loc = LeakLocator(KB, 'sensor_results')
    sensors = sorted(loc.sensor_nodes, key=lambda x: int(x))
    wn0 = wntr.network.WaterNetworkModel(INP)
    juncs = wn0.junction_name_list
    n2c = pickle.load(open(PART_PKL, 'rb'))[K]['node_to_community']
    xy = {n: np.array(wn0.get_node(n).coordinates) for n in juncs}

    r0 = wntr.sim.EpanetSimulator(copy.deepcopy(wn0)).run_sim(file_prefix=_P)
    p0 = r0.node['pressure'].mean(axis=0)
    d0 = r0.node['demand'].mean(axis=0)
    n_sims = 1

    def emitter_dp(node, target_Lps, alpha):
        """Emitter leak sized so the nominal discharge at baseline pressure is target_Lps."""
        nonlocal n_sims
        w = copy.deepcopy(wn0)
        w.options.hydraulic.emitter_exponent = alpha
        C = (target_Lps / 1000.0) / (max(float(p0[node]), 0.1) ** alpha)
        w.get_node(node).emitter_coefficient = C
        r = wntr.sim.EpanetSimulator(w).run_sim(file_prefix=_P)
        n_sims += 1
        pp = r.node['pressure'].mean(axis=0)
        realized = float((r.node['demand'].mean(axis=0)[node] - d0[node]) * 1000)
        return (pp - p0), realized

    def demand_dp(node, q_Lps):
        nonlocal n_sims
        w = copy.deepcopy(wn0)
        w.get_node(node).add_demand(q_Lps / 1000.0, '881')
        r = wntr.sim.EpanetSimulator(w).run_sim(file_prefix=_P)
        n_sims += 1
        return r.node['pressure'].mean(axis=0) - p0

    def vec(series):
        return np.array([float(series[s]) for s in sensors])

    # ================= Q1: is the fingerprint direction magnitude-invariant? =============
    rng = random.Random(SEED)
    probe = rng.sample(list(juncs), 20)
    print(f'\n[Q1] direction stability across magnitudes, {len(probe)} probe nodes')
    print(f'{"model":<26}{"min cos":>10}{"median cos":>12}{"mean cos":>10}{"worst pair":>28}')
    print('-' * 90)
    q1 = {}
    for label, fn in [('fixed demand (as published)', lambda nd, t: (demand_dp(nd, t), t)),
                      ('emitter alpha=0.5', lambda nd, t: emitter_dp(nd, t, 0.5)),
                      ('emitter alpha=1.0', lambda nd, t: emitter_dp(nd, t, 1.0)),
                      ('emitter alpha=1.5', lambda nd, t: emitter_dp(nd, t, 1.5))]:
        cosines, worst = [], (2.0, None)
        for nd in probe:
            V = []
            for t in INVAR_TARGETS:
                dp, _ = fn(nd, t)
                V.append(vec(dp))
            for i in range(len(V)):
                for j in range(i + 1, len(V)):
                    a, b = V[i], V[j]
                    na, nb = np.linalg.norm(a), np.linalg.norm(b)
                    if na == 0 or nb == 0:
                        continue
                    c = float(np.dot(a, b) / (na * nb))
                    cosines.append(c)
                    if c < worst[0]:
                        worst = (c, f'{nd}: {INVAR_TARGETS[i]:.0f} vs {INVAR_TARGETS[j]:.0f} L/s')
        cosines = np.array(cosines)
        q1[label] = dict(min=float(cosines.min()), median=float(np.median(cosines)),
                         mean=float(cosines.mean()))
        print(f'{label:<26}{cosines.min():>10.5f}{np.median(cosines):>12.5f}'
              f'{cosines.mean():>10.5f}{worst[1]:>28}')
    print('  (cos = 1.000 means the direction does not rotate with magnitude, so a '
          'scale-invariant\n   matcher covers the magnitude dimension and it need not be enumerated)')

    # ================= Q2: retrieval under model-form mismatch ==========================
    z = np.load(DENSE_CACHE, allow_pickle=True)
    dn = [str(x) for x in z['nodes']]
    Fd = z['F_day_normal']
    Fdn = np.linalg.norm(Fd, axis=1)
    print(f'\n[Q2] corpora: Cd = fixed-demand dense ({len(dn)} nodes, from exp2a)')

    t0 = time.time()
    Fe = np.zeros_like(Fd)
    for i, nd in enumerate(dn):
        dp, _ = emitter_dp(nd, Q_REF, 0.5)
        Fe[i] = vec(dp)
        if (i + 1) % 100 == 0:
            print(f'      building Ce (emitter alpha=0.5): {i+1}/{len(dn)} ({time.time()-t0:.0f}s)',
                  flush=True)
    Fen = np.linalg.norm(Fe, axis=1)
    print(f'      Ce built ({time.time()-t0:.0f}s)')

    held = {int(k): v for k, v in json.load(open('exp1b_results.json'))['config']['held_out_nodes'].items()}
    qs = [(c, n, t) for c, ns in held.items() for n in ns for t in TARGETS]
    random.Random(SEED).shuffle(qs)

    print(f'\n{"ground truth":<24}{"corpus":<26}{"part Top-1":>11}{"exact node":>12}'
          f'{"median m":>10}{"realized L/s":>14}')
    print('-' * 90)
    res = {}
    for alpha in ALPHAS:
        for cname, F, Fn in (('Cd fixed-demand', Fd, Fdn), ('Ce emitter a=0.5', Fe, Fen)):
            np.random.seed(SEED)
            ok, nok, dist, real = [], [], [], []
            for (gp, gn, t) in qs:
                dp, rq = emitter_dp(gn, t, alpha)
                real.append(rq)
                y = vec(dp) + np.random.normal(0, NOISE, len(sensors))
                r = proj_res_mat(y, F, Fn)
                i = int(np.argmin(r))
                pn = dn[i]
                ok.append(int(int(n2c[pn]) == gp))
                nok.append(int(pn == gn))
                dist.append(float(np.linalg.norm(xy[pn] - xy[gn])))
            res[f'a{alpha}_{cname[:2]}'] = dict(alpha=alpha, corpus=cname,
                                                part=float(np.mean(ok)), node=float(np.mean(nok)),
                                                med_m=float(np.median(dist)), ok=ok)
            print(f'{f"emitter alpha={alpha}":<24}{cname:<26}{100*np.mean(ok):>10.1f}%'
                  f'{100*np.mean(nok):>11.1f}%{np.median(dist):>9.0f}m{np.mean(real):>13.1f}',
                  flush=True)

    # fixed-demand ground truth, for reference (this is exp2a's setting)
    np.random.seed(SEED)
    ok = []
    for (gp, gn, t) in qs:
        y = vec(demand_dp(gn, t)) + np.random.normal(0, NOISE, len(sensors))
        i = int(np.argmin(proj_res_mat(y, Fd, Fdn)))
        ok.append(int(int(n2c[dn[i]]) == gp))
    print(f'{"fixed demand (control)":<24}{"Cd fixed-demand":<26}{100*np.mean(ok):>10.1f}%'
          f'{"":>11}{"":>10}{"":>14}')
    res['control'] = dict(part=float(np.mean(ok)), ok=ok)

    print('\npaired tests (same queries), Ce emitter corpus vs Cd fixed-demand corpus:')
    for alpha in ALPHAS:
        a = res[f'a{alpha}_Cd']['ok']
        b = res[f'a{alpha}_Ce']['ok']
        n01 = sum(1 for x, y_ in zip(a, b) if x == 0 and y_ == 1)
        n10 = sum(1 for x, y_ in zip(a, b) if x == 1 and y_ == 0)
        p = binomtest(n01, n01 + n10, 0.5).pvalue if n01 + n10 else 1.0
        print(f'  alpha={alpha}:  Cd {100*np.mean(a):5.1f}%  ->  Ce {100*np.mean(b):5.1f}%   '
              f'n01={n01} n10={n10}  p={p:.4g}')

    json.dump(dict(q1_direction_stability=q1, q2=res, n_simulations=n_sims,
                   probe_nodes=probe, targets=TARGETS, alphas=ALPHAS),
              open(OUT, 'w'), indent=1)
    print(f'\ntotal simulations {n_sims} | saved -> {OUT}')


if __name__ == '__main__':
    main()
